function buildOutfitCardDraftFromPrompt(prompt) {
  const tokens = tokenize(prompt) // remove acentos
  const detectedItems = []

  // Varre todas as opções de slot conhecidas
  for (const [slot, options] of
      Object.entries(OUTFIT_PIECE_OPTIONS)) {
    options.forEach(option => {
      // Busca por value, label e synonyms
      const found = [option.value,
        option.label, ...option.synonyms]
        .some(alias => tokens.includes(alias))
      if (found) detectedItems.push({
        piece_type: slot,
        display_label: option.label,
        color: normalizeDetectedColor(prompt),
        inferred_role: slot === 'upper'
          ? 'base_upper' : null,
        confidence: 0.5,
      })
    })
  }
  return { title, items: detectedItems, ... }
}

3 · Match AI → Guardaroba Real match
app/lib/outfit-ai-mapping.ts → mapAiInterpretationToManualForm()
function mapAiInterpretationToManualForm({
  interpretation, wardrobeItems }) {
  // Para cada item detectado pela AI...
  interpretation.items.forEach((item, i) => {
    const slot = normalizeDetectedPieceType(
      item.piece_type)

    // Tenta encontrar no guardaroba real
    const inventoryMatch = wardrobeItems.find(
      w => w.name.toLowerCase()
        .includes(item.display_label.toLowerCase()))

    // Fallback: opção sugerida pela AI
    const aiOptionId =
      `suggested:${slot}:ai-${i + 1}`

    // Prioriza peça real do guardaroba
    slotAssignments[slot] =
      inventoryMatch?.wardrobe_item_id
      || aiOptionId
  })
  return { slotAssignments, aiSlotOptions,
    style, occasion, title, mood }
}
