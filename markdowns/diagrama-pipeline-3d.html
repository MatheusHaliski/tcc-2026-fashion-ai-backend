def _skin_ratio(rgb):
  hsv = cv2.cvtColor(rgb, cv2.COLOR_RGB2HSV)
  # Hue 0–25: tons de pele humana
  lower = np.array([0, 35, 40])
  upper = np.array([25, 210, 255])
  mask = cv2.inRange(hsv, lower, upper)
  return np.count_nonzero(mask) / mask.size

def _face_confidence(gray):
  cascade = cv2.CascadeClassifier(
    "haarcascade_frontalface_default.xml")
  faces = cascade.detectMultiScale(gray,
    scaleFactor=1.1, minNeighbors=4)
  # Retorna 0.0 se nenhum rosto detectado

def _is_human_like(obj):
  # Detecta por tokens no nome do objeto
  _HUMAN_NAME_TOKENS = frozenset({
    "body","human","person","mannequin",
    "head","hand","foot","leg","arm",
    "hanger","rack","stand" ...
  })
  for token in _HUMAN_NAME_TOKENS:
    if token in obj.name.lower(): return True

  # Detecta por proporção altura/largura
  # Humano em pé: aspect ≈ 4-6
  # Camiseta: aspect ≈ 1.0–1.8
  height_z / width_xy >= 3.5 → humano

def _isolate_garment_mesh(mesh_objects):
  # Remove meshes humanos e stray props
  garment_candidates = [o for o in meshes
    if not _is_human_like(o)]
  # Remove props pequenos (botões, tags)
  poly_floor = max_polys * 0.05
  significant = [o for o in candidates
    if o.polygons >= poly_floor]
  # Mantém o maior mesh (a peça)
  garment = max(significant,
    key=lambda o: len(o.data.polygons))
  # Deleta tudo o mais da cena
  bpy.ops.object.delete()

def _extract_dominant_color(mat):
  # Amostra grade uniforme do centro (60%)
  # Quantiza em _COLOR_HIST_BUCKETS³ bins
  for row in range(y0, y1, stride):
    r,g,b = pixels[row*w + col : +4]
    key = (int(r*16), int(g*16), int(b*16))
    buckets[key] += (1, r, g, b)
  # Bin mais populoso = cor do tecido
  # (logos/manchas são minoria dos pixels)
  best = max(buckets, key=lambda v: v[0])
  return (best[1]/n, best[2]/n, best[3]/n)

def _purge_garment_materials(obj):
  dominant_rgb = _extract_dominant_color(
    obj.data.materials[0])
  obj.data.materials.clear()  # remove tudo
  # Cria material limpo SAI_Base_Fabric
  base_mat = bpy.data.materials.new(
    "SAI_Base_Fabric")
  bsdf = nodes.new("ShaderNodeBsdfPrincipled")
  bsdf.inputs["Base Color"].default_value = (
    *dominant_rgb, 1.0)
  bsdf.inputs["Roughness"].default_value = 0.80
  obj.data.materials.append(base_mat)

def _select_front_faces(obj, front_dir,
    dot_threshold=0.35):
  # Estágio 1: filtro por produto escalar
  for face in bm.faces:
    if face.normal.dot(local_front) >= 0.35:
      dot_selected.append(face)

  # Estágio 2: filtro de bounding-box do torso
  # Mantém apenas 60% central (horizontal)
  # e faixa 45–85% da altura (vertical)
  for face in dot_selected:
    centre = obj.matrix_world @ face.center
    if x_lo <= centre.x <= x_hi and \
       z_lo <= centre.z <= z_hi:
      torso_selected.append(face)

def _apply_decal_to_front_faces(
    obj, logo_path, front_dir, scale, offset):
  mat = _create_decal_material(logo_path)
  mat.blend_method = "BLEND"
  mat.use_backface_culling = True  # bloqueio traseiro
  # Atribui material às faces selecionadas
  for face in bm.faces:
    if face.select:
      face.material_index = mat_index
  # Projeta UV da vista frontal
  bpy.ops.uv.project_from_view(
    correct_aspect=True)
  # Aplica escala e offset vertical ao UV
  uv.x = (uv.x - 0.5) * logo_scale + 0.5
  uv.y = (uv.y - 0.5) * logo_scale + 0.5 + offset

def _analyze_back_preview_for_stains(path):
  # Carrega preview renderizado (RGBA float32)
  pixels = np.array(img.pixels[:]).reshape(h,w,4)
  garment_rgb = pixels[garment_mask]  # exclui fundo

  # Mediana robusta → cor real do tecido
  median_color = np.median(garment_rgb, axis=0)

  # Distância euclidiana de cada pixel à mediana
  distances = np.sqrt(((garment_rgb
    - median_color)**2).sum(axis=1))

  # > 0.20 em RGB normalizado = diferença visível
  hotspot_ratio = np.mean(distances > 0.20)

  # Score composto de artefatos
  artifact_confidence = min(1.0,
    hotspot_ratio * 3.5
    + max(0, color_variance - 0.08) * 2.5)

# Se confiança > 0.45 → re-sanitização agressiva
if artifact_confidence > 0.45:
  _sanitize_back_faces(garment, front_vec,
